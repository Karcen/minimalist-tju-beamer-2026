\section{学术内容}

\begin{frame}{公式排版}
  \begin{block}{带有 L2 惩罚的最小二乘目标}
    \[
      \hat{\bm\beta}
      =\arg\min_{\bm\beta}\left\{
        \frac{1}{n}\sum_{i=1}^{n}(y_i-\bm{x}_i^{\mathsf T}\bm\beta)^2
        +\lambda\lVert\bm\beta\rVert_2^2\right\}.
    \]
  \end{block}
  \medskip
  \begin{itemize}\setlength{\itemsep}{.65em}
    \item $\bm{x}_i$ 表示输入，$y_i$ 表示目标，$\lambda\geq0$ 控制惩罚强度。
    \item 用单独一行解释公式的含义，避免只展示符号。
  \end{itemize}
\end{frame}

\begin{frame}{定义、定理与命题}
  \begin{definition}[连续函数]
    函数 $f$ 在 $x_0$ 处连续，是指 $\lim_{x\to x_0}f(x)=f(x_0)$。
  \end{definition}
  \begin{theorem}[最值定理]
    若 $f$ 在闭区间 $[a,b]$ 上连续，则 $f$ 在该区间上取得最大值与最小值。
  \end{theorem}
  \begin{proposition}
    若 $f$ 在闭区间 $[a,b]$ 上连续，则 $f$ 在该区间上有界。
  \end{proposition}
\end{frame}
